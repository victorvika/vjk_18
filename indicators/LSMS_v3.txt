addLevel(price, side, kind, x1, col, w) =>
    if array.size(lvP) >= maxLvls
        line.delete(array.get(lvL, 0))
        array.shift(lvP)
        array.shift(lvT)
        array.shift(lvB)
        array.shift(lvK)
        array.shift(lvL)
    line ln = line.new(x1, price, bar_index, price, color = col, width = w, style = line.style_dashed)
    if not str.startswith(kind, "Swing")
        label.new(x1, price, kind, style = label.style_label_down, color = color.new(col, 100), textcolor = col, size = size.tiny)
    array.push(lvP, price)
    array.push(lvT, side)
    array.push(lvB, -1)
    array.push(lvK, kind)
    array.push(lvL, ln)
    ln

nearestLvl(side, px) =>
    float best = na
    int cnt = array.size(lvP)
    if cnt > 0
        for i = 0 to cnt - 1
            float p = array.get(lvP, i)
            if array.get(lvT, i) == side and array.get(lvB, i) == -1
                if side == 1 and p > px and (na(best) or p < best)
                    best := p
                else if side == -1 and p < px and (na(best) or p > best)
                    best := p
    best

hitBull(hi, lo) =>
    bool res = false
    int cnt = array.size(bullZ)
    if cnt > 0
        for i = 0 to cnt - 1
            box bz = array.get(bullZ, i)
            if hi >= box.get_bottom(bz) and lo <= box.get_top(bz)
                res := true
    res

hitBear(hi, lo) =>
    bool res = false
    int cnt = array.size(bearZ)
    if cnt > 0
        for i = 0 to cnt - 1
            box bz = array.get(bearZ, i)
            if hi >= box.get_bottom(bz) and lo <= box.get_top(bz)
                res := true
    res

clearSetup() =>
    while array.size(stpLn) > 0
        line.delete(array.pop(stpLn))
    while array.size(stpLb) > 0
        label.delete(array.pop(stpLb))
    while array.size(stpBx) > 0
        box.delete(array.pop(stpBx))
    true

drawSetup(isLong, entry, sl, tp1, tp2, zT, zB, zL, tag, rr1) =>
    clearSetup()
    color c = isLong ? bullCol : bearCol
    int x2 = bar_index + 15
    array.push(stpLn, line.new(bar_index, entry, x2, entry, color = color.blue, width = 1))
    array.push(stpLn, line.new(bar_index, sl, x2, sl, color = color.red, width = 1))
    array.push(stpLn, line.new(bar_index, tp2, x2, tp2, color = color.green, width = 1))
    array.push(stpLb, label.new(x2, entry, "Entry (" + tag + ") " + str.tostring(entry, format.mintick), style = label.style_label_left, color = color.blue, textcolor = color.white, size = size.tiny))
    array.push(stpLb, label.new(x2, sl, "SL " + str.tostring(sl, format.mintick), style = label.style_label_left, color = color.red, textcolor = color.white, size = size.tiny))
    array.push(stpLb, label.new(x2, tp2, "TP2 " + str.tostring(tp2, format.mintick) + " (" + str.tostring(rrTarget, "#.#") + "R)", style = label.style_label_left, color = color.green, textcolor = color.white, size = size.tiny))
    if not na(tp1)
        array.push(stpLn, line.new(bar_index, tp1, x2, tp1, color = color.green, width = 1, style = line.style_dotted))
        array.push(stpLb, label.new(x2, tp1, "TP1 liquidity " + str.tostring(tp1, format.mintick) + " (" + str.tostring(rr1, "#.#") + "R)", style = label.style_label_left, color = color.green, textcolor = color.white, size = size.tiny))
    if not na(zT)
        array.push(stpBx, box.new(zL, zT, x2, zB, border_color = color.new(c, 60), bgcolor = color.new(c, 85)))
    true

// ---- simulated outcome tracker (win = +rrTarget net of spread, loss = -1R) ----
vRec(idx, win) =>
    array.set(stN, idx, array.get(stN, idx) + 1)
    if win
        array.set(stW, idx, array.get(stW, idx) + 1)
    true

vAdd(side, en, sl, tp, bi, ses, knd) =>
    array.push(vT, float(side))
    array.push(vT, en)
    array.push(vT, sl)
    array.push(vT, tp)
    array.push(vT, float(bi))
    array.push(vT, 0.0)
    array.push(vT, float(ses))
    array.push(vT, float(knd))
    true

vResolve(hi, lo, bi) =>
    int n = math.floor(array.size(vT) / 8)
    if n > 0
        for i = n - 1 to 0
            int o = i * 8
            int sd = int(array.get(vT, o))
            float en = array.get(vT, o + 1)
            float sl = array.get(vT, o + 2)
            float tp = array.get(vT, o + 3)
            bool stopped = sd == 1 ? lo <= sl : hi >= sl
            bool target = sd == 1 ? hi >= tp : lo <= tp
            bool touched = sd == 1 ? lo <= en : hi >= en
            int res = 0
            if array.get(vT, o + 5) == 0
                if touched
                    array.set(vT, o + 5, 1.0)
                    res := stopped ? -1 : 0
                else if target or bi - int(array.get(vT, o + 4)) > fillBars
                    res := 2
            else
                res := stopped ? -1 : (target ? 1 : 0)
            if res != 0
                if res != 2
                    bool win = res == 1
                    vRec(0, win)
                    vRec(1 + int(array.get(vT, o + 6)), win)
                    vRec(3 + int(array.get(vT, o + 7)), win)
                    array.set(vDay, 0, array.get(vDay, 0) + (win ? rrTarget : -1.0))
                    if not win
                        array.set(vDay, 1, array.get(vDay, 1) + 1.0)
                for k = 1 to 8
                    array.remove(vT, o)
    true

kindIdx(k) =>
    str.startswith(k, "Asia") ? 0 : (str.startswith(k, "PD") ? 1 : (str.startswith(k, "EQ") ? 2 : 3))

bucketNeg(idx) =>
    int n = array.get(stN, idx)
    int w = array.get(stW, idx)
    n >= learnMin and (w * rrTarget - (n - w)) < 0

statTxt(idx) =>
    int n = array.get(stN, idx)
    int w = array.get(stW, idx)
    n == 0 ? "-" : str.tostring(n) + " | " + str.tostring(w * 100.0 / n, "#") + "% | " + str.tostring((w * rrTarget - (n - w)) / n, "#.##") + "R"
